/*JS*/
(function(){
  var btn=document.getElementById('mxBtn'),panel=document.getElementById('mxPanel'),close=document.getElementById('mxClose'),fab=document.getElementById('musicFab');
  var auto={run:false,dead:false};
  function stopAuto(){auto.dead=true;auto.run=false;}
  function open(){stopAuto();panel.classList.add('on');panel.setAttribute('aria-hidden','false');}
  function shut(){panel.classList.remove('on');panel.setAttribute('aria-hidden','true');}
  btn.addEventListener('click',open);close.addEventListener('click',shut);
  panel.querySelectorAll('a').forEach(function(a){a.addEventListener('click',function(e){e.preventDefault();shut();
    var t=document.querySelector(a.getAttribute('href'));if(t){var y=t.getBoundingClientRect().top+scrollY-10;window.scrollTo({top:y,behavior:'smooth'});}});});
  /* ── recorrido automático ── */
  var q=location.search+location.hash;
  var noAuto=/noauto/.test(q)||/#./.test(location.hash)||(window.matchMedia&&matchMedia('(prefers-reduced-motion: reduce)').matches);
  ['wheel','touchstart','pointerdown','keydown'].forEach(function(ev){window.addEventListener(ev,function(e){
    if(e.target&&e.target.closest&&e.target.closest('#mxBtn'))return; if(auto.run||!auto.dead)stopAuto();},{passive:true,capture:true});});
  var lastY=-1;
  window.addEventListener('scroll',function(){if(auto.run&&lastY>=0&&Math.abs(scrollY-lastY)>40)stopAuto();},{passive:true});
  function slowAt(){var y=scrollY+innerHeight*.5,ids=['hero','galeria','quote'];for(var i=0;i<ids.length;i++){var el=document.getElementById(ids[i]);if(!el)continue;var r=el.getBoundingClientRect();if(r.top<innerHeight*.5&&r.bottom>innerHeight*.5)return .6;}return 1;}
  function startAuto(){
    if(noAuto||auto.dead)return;auto.run=true;var acc=scrollY,t0=null,prev=null;
    function step(ts){
      if(!auto.run)return;
      if(document.hidden){prev=ts;requestAnimationFrame(step);return;}
      if(t0===null){t0=ts;prev=ts;}
      var dt=Math.min(64,ts-prev)/1000;prev=ts;
      var ease=Math.min(1,(ts-t0)/1000);var max=document.documentElement.scrollHeight-innerHeight;
      var rest=max-acc;if(rest<=1){auto.run=false;return;}
      var v=37*ease*slowAt()*Math.min(1,rest/40+.15);
      acc+=v*dt;lastY=Math.round(acc);window.scrollTo(0,acc);
      requestAnimationFrame(step);}
    requestAnimationFrame(step);}
  function shown(){btn.classList.add('on');setTimeout(startAuto,2500);}
  if(fab){if(fab.classList.contains('is-shown'))shown();else new MutationObserver(function(m,o){if(fab.classList.contains('is-shown')){o.disconnect();shown();}}).observe(fab,{attributes:true,attributeFilter:['class']});}
  else shown();
})();
