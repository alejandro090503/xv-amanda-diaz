(function(){
  var reduce=window.matchMedia&&matchMedia('(prefers-reduced-motion: reduce)').matches;
  document.querySelectorAll('canvas.px-dust').forEach(function(cv){
    var g=cv.getContext('2d'),dpr=Math.min(2,window.devicePixelRatio||1),W,H,P=[],on=false;
    var COL=['#B76E79','#D4AF37','#c98592','#E8C9A0'];
    function size(){var r=cv.getBoundingClientRect();W=r.width;H=r.height;if(!W||!H)return;cv.width=W*dpr;cv.height=H*dpr;g.setTransform(dpr,0,0,dpr,0,0);
      P=[];var n=Math.round(W*H/150);for(var i=0;i<n;i++)P.push({x:Math.random()*W,y:Math.pow(Math.random(),1.6)*H,r:Math.random()*1.1+.35,ph:Math.random()*6.3,sp:.5+Math.random()*2.4,vx:(Math.random()-.5)*.08,vy:-(.02+Math.random()*.07),c:COL[i%4]})}
    function f(t){g.clearRect(0,0,W,H);for(var i=0;i<P.length;i++){var p=P[i];if(!reduce){p.x+=p.vx;p.y+=p.vy;if(p.y<-2){p.y=H;p.x=Math.random()*W}}
        var a=.5+.5*Math.sin(t/1000*p.sp+p.ph);a=a*a;g.globalAlpha=a;g.fillStyle=p.c;g.beginPath();g.arc(p.x,p.y,p.r,0,6.3);g.fill();
        if(p.r>1.1&&a>.75){g.globalAlpha=(a-.75)*2.4;g.fillRect(p.x-3.2,p.y-.25,6.4,.5);g.fillRect(p.x-.25,p.y-3.2,.5,6.4)}}
      if(on&&!reduce)requestAnimationFrame(f)}
    size();addEventListener('resize',size);
    function inView(){var r=cv.getBoundingClientRect();return r.width>0&&r.bottom>0&&r.top<innerHeight}
    function kick(){var v=inView();if(v&&!on){size();on=true;requestAnimationFrame(f)}else if(!v)on=false}
    new IntersectionObserver(kick).observe(cv);addEventListener('scroll',kick,{passive:true});setInterval(kick,1200);
  });
})();
